cmake_minimum_required(VERSION 3.24)
project(VZipCli VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "vzip-cli currently targets Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(lzmasdk
  URL https://github.com/ip7z/7zip/releases/download/26.05/lzma2605.7z
  URL_HASH SHA256=668fda469b7f584f246008cf28359c66ef98d65fafb3b22a52672eebe6970c28
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(zstd
  URL https://codeload.github.com/facebook/zstd/tar.gz/refs/tags/v1.5.7
  URL_HASH SHA256=37d7284556b20954e56e1ca85b80226768902e2edabd3b649e9e72c0c9012ee3
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  SOURCE_SUBDIR build/cmake)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(lzmasdk zstd)
set(SDK "${lzmasdk_SOURCE_DIR}")
add_library(vzip_core STATIC src/VZip.cpp
  ${SDK}/C/LzmaDec.c ${SDK}/C/Alloc.c ${SDK}/C/CpuArch.c
  ${SDK}/C/7zCrc.c ${SDK}/C/7zCrcOpt.c)
target_include_directories(vzip_core PUBLIC src PRIVATE ${SDK}/C ${zstd_SOURCE_DIR}/lib)
target_link_libraries(vzip_core PUBLIC libzstd_static)
target_compile_definitions(vzip_core PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(vzip_plugin SHARED src/Plugin.cpp src/VZip.def)
set_target_properties(vzip_plugin PROPERTIES OUTPUT_NAME VZip)
target_include_directories(vzip_plugin PRIVATE ${SDK}/CPP)
target_link_libraries(vzip_plugin PRIVATE vzip_core oleaut32 ole32)
target_compile_definitions(vzip_plugin PRIVATE UNICODE _UNICODE)
add_executable(vzip src/Main.cpp)
target_link_libraries(vzip PRIVATE vzip_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(vzip_tests tests/CoreTests.cpp ${SDK}/C/LzmaEnc.c
    ${SDK}/C/LzFind.c ${SDK}/C/LzFindOpt.c)
  target_compile_definitions(vzip_tests PRIVATE Z7_ST)
  target_include_directories(vzip_tests PRIVATE ${SDK}/C ${zstd_SOURCE_DIR}/lib)
  target_link_libraries(vzip_tests PRIVATE vzip_core)
  add_test(NAME core COMMAND vzip_tests)
  add_executable(plugin_tests tests/PluginTests.cpp)
  target_include_directories(plugin_tests PRIVATE src ${SDK}/CPP)
  target_link_libraries(plugin_tests PRIVATE oleaut32 ole32)
  add_test(NAME plugin COMMAND plugin_tests $<TARGET_FILE:vzip_plugin>)
endif()
if(MSVC)
  foreach(target vzip_core vzip_plugin vzip)
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
  endforeach()
  if(BUILD_TESTING)
    foreach(target vzip_tests plugin_tests)
      target_compile_options(${target} PRIVATE /EHsc /utf-8)
    endforeach()
  endif()
  foreach(target vzip_core vzip_plugin vzip)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /guard:cf /sdl)
  endforeach()
  foreach(target vzip_plugin vzip)
    target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  endforeach()
endif()
